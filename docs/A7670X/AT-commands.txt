AT+CPIN?			# check the SIM state, if +CPIN: READY   (SIM unlocked)
					# if +CPIN: SIM PIN you need to enter the pin as below
AT+CPIN="1234"		# Replace 1234 with your real PIN, and keep the quotes. 
					# You should get OK, and AT+CPIN? should then show +CPIN: READY 

# you can disable the PIN lock:
AT+CLCK="SC",0,"1234"

# To turn it back on later, use:
AT+CLCK="SC",1,"1234"

	PHONEBOOK ON SIM

AT+CPBW				# to write a phonebook entry to a SIM card
					
# AT+CPBW=<index>,<number>,<type>,<text>   
# <index>: The memory location number (e.g., 1, 2).
# <number>: The phone number string.
# <type>: The type of number (e.g., 129 for international).
# <text>: The contact name in text format.

# Example:
# AT+CPBW=1,"+1234567890",129,"John Doe"

# Note that not all manufacturers implement the full AT+CPBW command set; some may require 
# AT+CPBS to select the storage area (e.g., SIM vs. Phone memory) before writing.
